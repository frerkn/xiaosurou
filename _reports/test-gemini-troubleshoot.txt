// 用法:
//   1. 打开 330, 进 chat 页面, F12 开 console
//   2. 复制整个文件 → 粘到 console → 回车
//   3. 进 chat 发 "帮我查下附近麦当劳"
//   4. 把 3 段 console 输出贴回来

(function () {
    console.log('=== 排查 1: 转换后 Gemini body 顶层 tools 字段 ===\n');

    if (!window.McpBridge || !McpBridge.debugGeminiToolLoop) {
        console.error('❌ McpBridge.debugGeminiToolLoop 不存在 (加载 v0.1.76 之前版本?)');
        return;
    }
    const fakeUrl = 'https://generativelanguage.googleapis.com/v1beta/models/gemini-3.6-pro:generateContent?key=AIza_FAKE';
    const fakeBody = JSON.stringify({
        model: 'gemini-3.6-pro',
        messages: [
            { role: 'system', content: '你是助手' },
            { role: 'user', content: '帮我查下附近麦当劳' }
        ]
    });
    const r = McpBridge.debugGeminiToolLoop({ url: fakeUrl, body: fakeBody });
    if (r.error) {
        console.error('❌ 转换失败:', r.error);
        return;
    }
    console.log('exposed 工具数:', r.exposedToolsCount);
    console.log('exposed 工具名:', r.exposedToolNames);
    console.log('\n--- 转换后 tools 顶层结构 (前 1 个工具) ---');
    if (r.geminiBody.tools && r.geminiBody.tools[0]) {
        const decl0 = r.geminiBody.tools[0].functionDeclarations[0];
        console.log(JSON.stringify({
            toolsIsArray: Array.isArray(r.geminiBody.tools),
            toolsLength: r.geminiBody.tools.length,
            tools0Keys: Object.keys(r.geminiBody.tools[0]),
            hasFunctionDeclarations: 'functionDeclarations' in r.geminiBody.tools[0],
            functionDeclarationsLength: r.geminiBody.tools[0].functionDeclarations?.length,
            firstDecl: { name: decl0.name, description: decl0.description?.slice(0, 80), parametersType: decl0.parameters?.type }
        }, null, 2));
    } else {
        console.log('❌ tools 顶层结构异常, 完整 dump:');
        console.log(JSON.stringify(r.geminiBody.tools, null, 2));
    }

    console.log('\n\n=== 排查 3: schema type 是不是全大写 ===\n');
    if (r.geminiBody.tools?.[0]?.functionDeclarations) {
        const decl = r.geminiBody.tools[0].functionDeclarations[0];
        console.log('第 1 个工具:', decl.name);
        console.log('顶层 parameters.type:', decl.parameters?.type, '(期望 OBJECT)');
        const lowercaseTypes = [];
        const allTypes = [];
        function scan(node, path) {
            if (!node) return;
            if (node.type) {
                allTypes.push({ path, type: node.type });
                if (node.type === node.type.toLowerCase() && node.type !== 'type') {
                    lowercaseTypes.push({ path, type: node.type });
                }
            }
            if (node.properties) for (const k in node.properties) scan(node.properties[k], path + '.' + k);
            if (node.items) scan(node.items, path + '.items');
        }
        for (const d of r.geminiBody.tools[0].functionDeclarations) scan(d.parameters, d.name);
        if (lowercaseTypes.length) {
            console.log('❌ 发现小写 type (' + lowercaseTypes.length + ' 个):');
            console.log(lowercaseTypes);
        } else {
            console.log('✅ 所有 type 合法, 找到的 type 值:', [...new Set(allTypes.map(t => t.type))].join(','));
        }
    } else {
        console.log('⚠️ 跳过 (没工具)');
    }

    console.log('\n\n=== 排查 2: 真请求是不是进 runChatWithToolLoopGemini ===\n');
    console.log('现在去 chat 发 "帮我查下附近麦当劳", 触发后看下面 4 行:');
    console.log('  - "[排查2] 真实 fetch Gemini..." → 看到 = 进 fetch hook');
    console.log('  - "[McpBridge] Gemini 已合并 N 个 MCP 工具" → 看到 = 进新循环');
    console.log('  - "[McpBridge] tool_start" / "tool_ok" → 看到 = 工具真在调');
    console.log('  - 啥都没打 = 没进新循环, 走 originalFetch bypass 了\n');

    const orig = window.fetch;
    let hookedFetchCalled = 0;
    window.fetch = function (input, init) {
        const u = typeof input === 'string' ? input : (input && input.url) || '';
        if (u.includes('generativelanguage.googleapis.com')) {
            hookedFetchCalled++;
            const bodyObj = init && init.body ? (() => { try { return JSON.parse(init.body); } catch (e) { return null; } })() : null;
            console.log('[排查2] 真实 fetch Gemini, 次数:', hookedFetchCalled);
            console.log('  url:', u.slice(0, 100));
            console.log('  body.model:', bodyObj?.model);
            console.log('  body.tools 顶层:', Array.isArray(bodyObj?.tools) ? '数组 长度=' + bodyObj.tools.length : typeof bodyObj?.tools);
            if (bodyObj?.tools?.[0]) {
                console.log('  body.tools[0] keys:', Object.keys(bodyObj.tools[0]));
                console.log('  body.tools[0].functionDeclarations 长度:', bodyObj.tools[0].functionDeclarations?.length);
            }
        }
        return orig.apply(this, arguments);
    };
    console.log('✅ fetch hook 已装, 现在发 "帮我查下附近麦当劳"\n');
})();
