// ============================================================
// Gemini 调工具 - 浏览器 console 验证脚本 (v0.1.76c)
// ============================================================
// 1. 打开 330, 进 chat 页面, F12 开 console
// 2. 把整个文件复制到 console, 跑
// 3. 看 console 输出对不对
// 4. 出问题把 console 截图 + 这个 snippet 输出贴回来
// ============================================================

(async function () {
    console.log('=== Gemini 调工具验证 ===\n');

    // ===== 1. 基础检查 =====
    console.log('[1/4] 基础检查');
    console.log('  McpBridge 已加载:', !!window.McpBridge);
    console.log('  McpGenericClient 已加载:', !!window.McpGenericClient);
    console.log('  Hook 已安装:', McpBridge.isHookInstalled());
    console.log('  已启用 server 数:', McpBridge.getEnabledServerCount());
    console.log('  已启用 server 列表:', McpBridge.getEnabledServers().map(s => s.name + '(' + (s.tools || []).length + ')').join(', '));
    console.log('');

    if (McpBridge.getEnabledServerCount() === 0) {
        console.warn('⚠️ 没有 enabled MCP server, 先去设置开一个 (如 mcd) 再跑');
        return;
    }

    // ===== 2. 看 Gemini body 转换对不对 (不发请求) =====
    console.log('[2/4] Gemini body 转换验证 (不发请求)');
    const fakeUrl = 'https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=AIza_FAKE';
    const fakeBody = JSON.stringify({
        model: 'gemini-2.0-flash',
        messages: [
            { role: 'system', content: '你是助手, 用角色语气回复' },
            { role: 'user', content: '帮我查下附近麦当劳' },
            { role: 'assistant', content: '好, 我查查' }
        ]
    });
    const debugResult = McpBridge.debugGeminiToolLoop({ url: fakeUrl, body: fakeBody });
    if (debugResult.error) {
        console.error('  ❌ 转换失败:', debugResult.error);
        return;
    }
    console.log('  ✅ 转换成功');
    console.log('  exposed 工具数:', debugResult.exposedToolsCount);
    console.log('  exposed 工具名前 5:', debugResult.exposedToolNames.slice(0, 5).join(', '));
    console.log('  Gemini body 大小:', debugResult.geminiBodySize, '字符 (一般 5K-15K, 太大 > 30K 可能超 Gemini systemInstruction 8K token 限制)');
    console.log('  systemInstruction 前 200 字:');
    console.log('    ', debugResult.geminiBody.systemInstruction.parts[0].text.slice(0, 200).replace(/\n/g, '\n     '));
    console.log('  contents 长度:', debugResult.geminiBody.contents.length);
    console.log('  contents[0] (user message):', JSON.stringify(debugResult.geminiBody.contents[0]));
    console.log('  contents[2] (model message):', JSON.stringify(debugResult.geminiBody.contents[2]));
    console.log('');

    // ===== 3. 看 tools 格式 (第 1 个工具) =====
    console.log('[3/4] tools 格式验证 (前 1 个工具)');
    const firstTool = debugResult.geminiBody.tools[0].functionDeclarations[0];
    console.log('  name:', firstTool.name);
    console.log('  description 前 100 字:', firstTool.description.slice(0, 100));
    console.log('  parameters.type (应=OBJECT):', firstTool.parameters.type);
    if (firstTool.parameters.properties) {
        const propNames = Object.keys(firstTool.parameters.properties);
        console.log('  参数列表:', propNames.join(', '));
        // 找带 enum 的参数 (mcd query-nearby-stores 的 beType 这种)
        for (const k of propNames) {
            const p = firstTool.parameters.properties[k];
            if (p.enum) {
                console.log('    参数 ' + k + ' enum:', JSON.stringify(p.enum), '(必须是 ["1","5"] 这种 string 数组)');
            }
        }
    }
    console.log('');

    // ===== 4. 跑一次真请求 (需要你填 key) =====
    console.log('[4/4] 真请求测试 (需要 Gemini API key)');
    console.log('  跳过 (这步要在 chat 页面用真 key 触发, 跟 AI 说"帮我查下附近麦当劳")');
    console.log('');
    console.log('  真请求测试步骤:');
    console.log('  a) 打开 330 设置 → API 设置 → 选 Gemini native 渠道 (不是 OpenAI 兼容)');
    console.log('  b) 确认 URL 是 https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=AIza... 这种');
    console.log('  c) 进任一 chat, 发 "帮我查下附近麦当劳"');
    console.log('  d) 看 console:');
    console.log('     - 看到 "[McpBridge] Gemini 已合并 N 个 MCP 工具" → 转换层接住了');
    console.log('     - 看到 "[McpBridge] tool_start" / "tool_ok" → 工具调通了');
    console.log('     - 看到 "[McpBridge] Gemini 工具循环异常" → 有 bug, 把错误信息贴回来');
    console.log('     - 看到 "[McpBridge] AI 已完成" + AI 文本里提到门店 → 全链路 work');
    console.log('');
    console.log('  e) 出问题时把这 4 段 console 输出 + 出错截图贴回来');

    console.log('\n=== 验证完成 ===');
})().catch(e => console.error('脚本异常:', e));
